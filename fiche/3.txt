def construire_nom_sortie(nom_base, index):

3. Gestion des conflits
Si le fichier existe déjà, ajouter un suffixe _1, _2, etc.

python
def gerer_conflit(dossier, nom_fichier):
    """Vérifie si le fichier existe et ajoute un suffixe si nécessaire."""
    chemin = dossier / nom_fichier
    if not chemin.exists():
        return chemin
    
    # Ajouter un suffixe
    base = chemin.stem
    suffixe = 1
    while True:
        nouveau_nom = f"{base}_{suffixe}{chemin.suffix}"
        nouveau_chemin = dossier / nouveau_nom
        if not nouveau_chemin.exists():
            return nouveau_chemin
        suffixe += 1
4. Commande FFmpeg
La commande conceptuelle :

bash
ffmpeg -i video_source.mp4 -i audio_source.mp4 \
       -c:v copy -map 0:v:0 -map 1:a:0 -shortest output.mp4
Explication :

-i video_source.mp4 : premier fichier (on prend la vidéo)

-i audio_source.mp4 : deuxième fichier (on prend l'audio)

-c:v copy : copier la vidéo sans ré-encodage (rapide)

-map 0:v:0 : prendre la première piste vidéo du premier fichier

-map 1:a:0 : prendre la première piste audio du deuxième fichier

-shortest : couper à la durée la plus courte (sécurité)

5. Lancement de FFmpeg
python
def remplacer_audio(fichier_video, fichier_audio, fichier_sortie):
    """Remplace l'audio d'un fichier vidéo."""
    cmd = [
        "ffmpeg",
        "-i", fichier_video,
        "-i", fichier_audio,
        "-c:v", "copy",
        "-map", "0:v:0",
        "-map", "1:a:0",
        "-shortest",
        "-y",  # Écraser sans demander
        str(fichier_sortie)
    ]
    
    try:
        resultat = sub.run(cmd, capture_output=True, text=True)
        if resultat.returncode != 0:
            print(f"❌ Erreur FFmpeg : {resultat.stderr}")
            return False
        return True
    except Exception as e:
        print(f"❌ Erreur : {e}")
        return False
Note : Le paramètre -y force l'écrasement. Comme on gère déjà les conflits de nom, on peut l'utiliser sans risque.

6. Traitement par lot
python
def traiter_paires(paires, nom_base):
    """Traite toutes les paires et retourne le nombre de réussites/échecs."""
    dossier_sortie = creer_dossier_sortie()
    reussites = 0
    echecs = 0
    
    for i, (audio, video, duree_audio, duree_video) in enumerate(paires, start=1):
        # Construire le nom de sortie
        nom_fichier = construire_nom_sortie(nom_base, i)
        chemin_sortie = gerer_conflit(dossier_sortie, nom_fichier)
        
        print(f"Traitement de la paire {i} : {video} + {audio} → {chemin_sortie.name}")
        
        # Lancer le remplacement
        if remplacer_audio(video, audio, chemin_sortie):
            reussites += 1
            print(f"  ✅ Réussi")
        else:
            echecs += 1
            print(f"  ❌ Échec")
    
    return reussites, echecs
